// ── 1. Add these DbSet properties inside AppDbContext class ──────

// Module 20 – Coupons
public DbSet<Coupon>      Coupons      { get; set; }
public DbSet<CouponUsage> CouponUsages { get; set; }

// Module 24 – Live Tracking
public DbSet<BookingTracking> BookingTrackings { get; set; }


// ── 2. Add inside OnModelCreating(ModelBuilder mb) ───────────────

mb.Entity<Coupon>()
    .HasIndex(c => c.Code)
    .IsUnique();

mb.Entity<Coupon>()
    .HasIndex(c => c.IsActive);

mb.Entity<CouponUsage>()
    .HasIndex(u => new { u.CouponId, u.UserId });

mb.Entity<BookingTracking>()
    .HasIndex(t => t.BookingId)
    .IsUnique();  // one tracking record per booking

mb.Entity<BookingTracking>()
    .HasIndex(t => t.ChefId);
