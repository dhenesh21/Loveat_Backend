// ── 1. Add these DbSet properties inside AppDbContext class ──────

// Module 31 – Support
public DbSet<SupportTicket>  SupportTickets  { get; set; }
public DbSet<SupportMessage> SupportMessages { get; set; }
public DbSet<FAQ>            FAQs            { get; set; }

// Module 32 – Campaigns
public DbSet<NotificationCampaign> NotificationCampaigns { get; set; }


// ── 2. Add inside OnModelCreating(ModelBuilder mb) ───────────────

mb.Entity<SupportTicket>()
    .HasIndex(t => t.UserId);

mb.Entity<SupportTicket>()
    .HasIndex(t => t.Status);

mb.Entity<SupportTicket>()
    .HasIndex(t => t.TicketNumber)
    .IsUnique();

mb.Entity<SupportMessage>()
    .HasIndex(m => m.TicketId);

mb.Entity<FAQ>()
    .HasIndex(f => new { f.Category, f.Audience });

mb.Entity<NotificationCampaign>()
    .HasIndex(c => c.Status);

mb.Entity<NotificationCampaign>()
    .HasIndex(c => c.ScheduledAt);
